public class IntsRefBuilder {
    private int[] refs;
    private int refCount;

    public IntsRefBuilder() {
        refs = new int[0];
        refCount = 0;
    }

    public IntsRefBuilder(int[] refs) {
        this.refs = refs;
        refCount = refs.length;
    }

    public int[] getRefs() {
        return refs;
    }

    public int getRefCount() {
        return refCount;
    }

    public void addRef(int ref) {
        refs[refCount++] = ref;
    }

    public int getRef(int index) {
        return refs[index];
    }

    public int getRefCount() {
        return refCount;
